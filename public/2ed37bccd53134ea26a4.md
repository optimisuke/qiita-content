## コード
必要最小限（たぶん）で書いてみた。

```bash:test.sh
#!/bin/bash

# 環境変数
apikey=$WA_API_KEY
url=$WA_URL

# environment_idを取得
assistant=$(curl -X GET -u "apikey:${apikey}" "${url}/v2/assistants?version=2021-11-27")
environment_id=$(echo $assistant | jq -r '.assistants[0].assistant_environments[0].environment_id')
echo $environment_id

# session_idを取得
res_sessions=$(curl -X POST -u "apikey:${apikey}" "${url}/v2/assistants/${environment_id}/sessions?version=2021-11-27")
session_id=$(echo $res_sessions | jq -r '.session_id')
echo $session_id

# メッセージ送信
input_text="こんにちは"
message=$(curl -X POST -u "apikey:${apikey}" --header "Content-Type:application/json" --data "{\"input\": {\"text\": \"${input_text}\"}}" "${url}/v2/assistants/${environment_id}/sessions/${session_id}/message?version=2021-11-27")
text=$(echo $message | jq -r '.output.generic[0].text')
echo $text

# セッション削除
curl -X DELETE -u "apikey:${apikey}" "${url}/v2/assistants/${environment_id}/sessions/${session_id}?version=2021-11-27"
```
