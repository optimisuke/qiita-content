# はじめに
PythonとUnittestの勉強中。
ためしに、Class作って、Unittestしてみた。
selfだらけで、自信ない。
どなたかコメントもらえるとうれしいです。

# コード
jsonのデータを取得する部分とそこから温度、湿度を取り出す部分とに分けてみた。
オブジェクトを生成して、違うタイミングでまたデータをとることも想定して、jsonデータの取得は`__init__`ではやらないことにした。なんとなく都市名で初期化してみた。

```python:openweather.py
import requests
import json

class OpenWeatherMap():
    """Get weather from OpenWeatherMap API                                           
    """

    def __init__(self,city):
        """Initialize city                                                           
        """
        self.city = city
        self.api_key = "xxx"
        self.api = "http://api.openweathermap.org/data/2.5/weather?units=metric&q={c\
ity}&APPID={key}"
        self.url = self.api.format(city = self.city, key = self.api_key)

    def get_json_weather(self):
        """Get json weather data                                                     
        """
        response = requests.get(self.url)
        data = response.json()
        # jsonText = json.dumps(data, indent=4)                                      
        # print(jsonText)                                                            
        return data

    def get_temp(self, data):
        """Get temperature data from json data                                       
        """
        temp = data['main']['temp']
        return temp

    def get_humi(self, data):
        """Get Humidity data from json data                                          
        """
        humi = data['main']['humidity']
        return humi


if __name__ == '__main__':
    owm_kobe = OpenWeatherMap("Kobe")
    json_weather = owm_kobe.get_json_weather()
    print(owm_kobe.get_temp(json_weather))
    print(owm_kobe.get_humi(json_weather))
```


# テスト
何をテストすべきか悩ましいけど、それっぽい温度になるかだけ見ることにした。APIでちゃんとアクセスできるかはまた別途テストすべきなんかもしれない。テストというよりもエラー処理すべきなのかもしれない。よくわからん。

```python:tests/test_openweather.py
import unittest
import openweather

class TestCalc(unittest.TestCase):
    def setUp(self):
        self.owm_kobe = openweather.OpenWeatherMap("Kobe")
        self.json_weather = self.owm_kobe.get_json_weather()

    def test_get_temp(self):
        temp = self.owm_kobe.get_temp(self.json_weather)
        print(temp)
        self.assertTrue(temp < +50)
        self.assertTrue(temp > -10)

    def test_get_humi(self):
        humi = self.owm_kobe.get_humi(self.json_weather)
        print(humi)
        self.assertTrue(humi < 100)
        self.assertTrue(humi > 0)

    def tearDown(self):
        del self.owm_kobe
```
